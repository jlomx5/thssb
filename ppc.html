/** * CONVECSA ERP 360° - SISTEMA RDO LEAN THSSB * Enrutador automático de hojas: * - REGISTRO_DIARIO_PINTURA * - REGISTRO_DIARIO_SCF */ function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(15000); const ss = SpreadsheetApp.getActiveSpreadsheet(); const data = JSON.parse(e.postData.contents); // 1. Identificar la especialidad (PINTURA o SCF) const modulo = (data.modulo || "PINTURA").toUpperCase(); const nombreHoja = modulo === "SCF" ? "REGISTRO_DIARIO_SCF" : "REGISTRO_DIARIO_PINTURA"; // Si existe la hoja antigua "REGISTRO_DIARIO" y estamos en PINTURA, renombrarla automáticamente let sheet = ss.getSheetByName(nombreHoja); if (!sheet) { const oldSheet = ss.getSheetByName("REGISTRO_DIARIO"); if (oldSheet && modulo === "PINTURA") { oldSheet.setName("REGISTRO_DIARIO_PINTURA"); sheet = oldSheet; } else { sheet = ss.insertSheet(nombreHoja); sheet.appendRow([ "ID_REGISTRO", "TIMESTAMP", "FECHA_OBRA", "MODULO", "USUARIO_REGISTRA", "OPERARIO_CAMPO", "NIVEL_THSSB", "ELEMENTO", "SUBPROCESO", "METRADO_EJECUTADO", "UNIDAD", "HH_REALES", "REND_OPTIMO", "HH_GANADAS", "EFICIENCIA_PCT", "ESTADO_LEAN", "RESTRICCION", "OBSERVACION" ]); sheet.getRange("A1:R1").setBackground("#0052CC").setFontColor("#FFFFFF").setFontWeight("bold"); sheet.setFrozenRows(1); } } // Asegurar que también exista la hoja de SCF aunque aún no tenga registros if (!ss.getSheetByName("REGISTRO_DIARIO_SCF")) { const shSCF = ss.insertSheet("REGISTRO_DIARIO_SCF"); shSCF.appendRow([ "ID_REGISTRO", "TIMESTAMP", "FECHA_OBRA", "MODULO", "USUARIO_REGISTRA", "OPERARIO_CAMPO", "NIVEL_THSSB", "ELEMENTO", "SUBPROCESO", "METRADO_EJECUTADO", "UNIDAD", "HH_REALES", "REND_OPTIMO", "HH_GANADAS", "EFICIENCIA_PCT", "ESTADO_LEAN", "RESTRICCION", "OBSERVACION" ]); shSCF.getRange("A1:R1").setBackground("#0052CC").setFontColor("#FFFFFF").setFontWeight("bold"); shSCF.setFrozenRows(1); } // 2. Cálculos Lean (Base cuadrilla = 17 HH) const timestampExacto = Utilities.formatDate(new Date(), "GMT-5", "yyyy-MM-dd HH:mm:ss"); const fechaObra = data.fecha || Utilities.formatDate(new Date(), "GMT-5", "yyyy-MM-dd"); const metrado = Number(data.metrado) || 0.0; const hhReal = Number(data.hhReal) || 1.0; const rendOpt = Number(data.rendOpt) || (modulo === "SCF" ? 43.0 : 52.0); const hhGanadas = metrado * (17.0 / rendOpt); const ef = hhReal > 0 ? (hhGanadas / hhReal) * 100 : 0; const estado = ef >= 100 ? "GANANCIA" : "PERDIDA"; const lastRow = sheet.getLastRow() + 1; const prefijo = modulo === "SCF" ? "SCF-" : "PIN-"; const idRegistro = prefijo + Utilities.formatDate(new Date(), "GMT-5", "yyyyMMdd") + "-" + lastRow; const unidad = modulo === "PINTURA" ? "m²" : (String(data.subproceso).includes("ml") || data.elemento === "JUNTAS" ? "ml" : "und"); sheet.appendRow([ idRegistro, timestampExacto, fechaObra, modulo, data.usuarioEmail || "jorosco@convecsa.pe", data.trabajador || "No asignado", data.nivel || "P1", data.elemento || "MURO", data.subproceso || "", metrado, unidad, hhReal, rendOpt, Number(hhGanadas.toFixed(2)), Number((ef / 100).toFixed(4)), estado, data.causa || "Ninguna", data.detalle || "" ]); SpreadsheetApp.flush(); lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({ status: "ok", hoja: nombreHoja, id: idRegistro, estado: estado, eficiencia: ef.toFixed(1) })).setMimeType(ContentService.MimeType.JSON); } catch (err) { lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const hojas = ["REGISTRO_DIARIO_PINTURA", "REGISTRO_DIARIO_SCF", "REGISTRO_DIARIO"]; let todos = []; hojas.forEach(nombre => { const sh = ss.getSheetByName(nombre); if (!sh) return; const rows = sh.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { const r = rows[i]; if (!r[0]) continue; const fechaFila = r[2] instanceof Date ? Utilities.formatDate(r[2], "GMT-5", "yyyy-MM-dd") : String(r[2]).split("T")[0]; todos.push({ id: r[0], horaExacta: String(r[1]), fecha: fechaFila, modulo: r[3] || (nombre.includes("SCF") ? "SCF" : "PINTURA"), hoja: nombre, usuario: r[4], trabajador: r[5], nivel: r[6], elemento: r[7], sub: r[8], metrado: Number(r[9]), unidad: r[10] || "m²", hhReal: Number(r[11]), rendOpt: Number(r[12]), hhGan: Number(r[13]), ef: (Number(r[14]) * 100).toFixed(1), est: r[15] || (Number(r[14]) >= 1 ? "GANANCIA" : "PERDIDA"), restr: r[16], obs: r[17] }); } }); return ContentService.createTextOutput(JSON.stringify({ registros: todos })) .setMimeType(ContentService.MimeType.JSON); }
