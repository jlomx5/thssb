/** * CONVECSA ERP 360° - BACKEND MULTI-HOJA THSSB * Enrutamiento automático a: * 1. REGISTRO_DIARIO_PINTURA * 2. REGISTRO_DIARIO_SCF */ function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(15000); const ss = SpreadsheetApp.getActiveSpreadsheet(); // Parseo de los datos enviados desde ppc.html const data = JSON.parse(e.postData.contents); const modulo = (data.modulo || "PINTURA").toUpperCase(); // 1. Determinar el nombre de la hoja según la partida const nombreHoja = modulo === "SCF" ? "REGISTRO_DIARIO_SCF" : "REGISTRO_DIARIO_PINTURA"; let sheet = ss.getSheetByName(nombreHoja); // Si la hoja no existe, se crea con sus cabeceras corporativas CONVECSA if (!sheet) { sheet = ss.insertSheet(nombreHoja); sheet.appendRow([ "ID_REGISTRO", "TIMESTAMP_EXACTO", "FECHA_OBRA", "PARTIDA", "USUARIO_REGISTRA", "OPERARIO_CAMPO", "NIVEL_THSSB", "ELEMENTO", "SUBPROCESO", "METRADO_EJECUTADO", "UNIDAD", "HH_REALES", "REND_OPTIMO_TARGET", "HH_GANADAS", "EFICIENCIA_PCT", "ESTADO_LEAN", "RESTRICCION_CAUSA", "DETALLE_UBICACION" ]); sheet.getRange("A1:R1").setBackground("#0052CC").setFontColor("#FFFFFF").setFontWeight("bold"); sheet.setFrozenRows(1); } // 2. Extracción y normalización de variables const timestampExacto = Utilities.formatDate(new Date(), "GMT-5", "yyyy-MM-dd HH:mm:ss"); const fechaObra = data.fecha || Utilities.formatDate(new Date(), "GMT-5", "yyyy-MM-dd"); const usuarioEmail = data.usuarioEmail || "jorosco@convecsa.pe"; const operario = data.trabajador || "No asignado"; const nivel = data.nivel || "P1"; const elemento = data.elemento || "MURO"; const subproceso = data.subproceso || ""; const metrado = Number(data.metrado) || 0.0; const hhReal = Number(data.hhReal) || 1.0; const rendOpt = Number(data.rendOpt) || (modulo === "SCF" ? 43.0 : 52.0); const causa = data.causa || "Ninguna"; const obs = data.detalle || ""; // Base de cálculo Lean (Cuadrilla estándar = 17 HH por jornada) const hhGanadas = metrado * (17.0 / rendOpt); const eficiencia = hhReal > 0 ? (hhGanadas / hhReal) * 100 : 0; const estado = eficiencia >= 100 ? "GANANCIA" : "PERDIDA"; const lastRow = sheet.getLastRow() + 1; const idRegistro = (modulo === "SCF" ? "SCF-" : "PIN-") + Utilities.formatDate(new Date(), "GMT-5", "yyyyMMdd") + "-" + lastRow; const unidad = modulo === "PINTURA" ? "m²" : (subproceso.includes("ml") || elemento === "JUNTAS" ? "ml" : "und"); // 3. Inserción de fila en la hoja correspondiente sheet.appendRow([ idRegistro, timestampExacto, fechaObra, modulo, usuarioEmail, operario, nivel, elemento, subproceso, metrado, unidad, hhReal, rendOpt, Number(hhGanadas.toFixed(2)), Number((eficiencia / 100).toFixed(4)), estado, causa, obs ]); SpreadsheetApp.flush(); lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({ status: "ok", hojaDestino: nombreHoja, id: idRegistro, timestamp: timestampExacto })).setMimeType(ContentService.MimeType.JSON); } catch (err) { lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } // LECTURA CONSOLIDADA PARA EL DASHBOARD (GET) function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const hojas = ["REGISTRO_DIARIO_PINTURA", "REGISTRO_DIARIO_SCF"]; let todosLosRegistros = []; hojas.forEach(nombre => { const sh = ss.getSheetByName(nombre); if (!sh) return; const rows = sh.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { const r = rows[i]; if (!r[0]) continue; const fechaFila = r[2] instanceof Date ? Utilities.formatDate(r[2], "GMT-5", "yyyy-MM-dd") : String(r[2]).split("T")[0]; todosLosRegistros.push({ id: r[0], horaExacta: r[1], fecha: fechaFila, modulo: r[3], usuario: r[4], rol: String(r[4]).includes("jorosco") ? "SUPER_ADMIN" : "CAPATAZ", trabajador: r[5], nivel: r[6], elemento: r[7], sub: r[8], metrado: Number(r[9]), unidad: r[10], hhReal: Number(r[11]), rendOpt: Number(r[12]), hhGan: Number(r[13]), ef: (Number(r[14]) * 100).toFixed(1), est: r[15], restr: r[16], obs: r[17] }); } }); // Ordenar cronológicamente descendente (los más recientes primero) todosLosRegistros.sort((a, b) => new Date(b.horaExacta) - new Date(a.horaExacta)); return ContentService.createTextOutput(JSON.stringify({ registros: todosLosRegistros })) .setMimeType(ContentService.MimeType.JSON); }
